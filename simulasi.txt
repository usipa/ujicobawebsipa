    <script>
        document.getElementById('simulasiForm').addEventListener('submit', function (e) {
            e.preventDefault();

            const pinjaman = parseFloat(document.getElementById('jumlahPinjaman').value);
            const totalCicilan = parseInt(document.getElementById('totalCicilan').value);
            const bulanMulaiValue = document.getElementById('bulanMulai').value;

            // Bunga Flat rate: 0.75% per bulan
            const BUNGA_PERSEN_BULAN = 0.0075; 
            const MAX_PINJAMAN = 20000000;

            if (isNaN(pinjaman) || isNaN(totalCicilan) || pinjaman <= 0 || totalCicilan <= 0 || pinjaman > MAX_PINJAMAN) {
                alert(`Mohon periksa input Anda. Jumlah pinjaman maksimal adalah ${formatRupiah(MAX_PINJAMAN)}.`);
                return;
            }

            // --- Perhitungan ---
            
            // 1. Angsuran Pokok Bulanan
            const pokokBulanan = pinjaman / totalCicilan;

            // 2. Bunga Bulanan (0.75% * Pokok Pinjaman Awal)
            const bungaBulanan = pinjaman * BUNGA_PERSEN_BULAN;

            // 3. Total Angsuran Bulanan
            const totalAngsuranBulanan = pokokBulanan + bungaBulanan;
            
            // 4. Inisialisasi Tabel
            const tabelBody = document.getElementById('tabelBody');
            tabelBody.innerHTML = ''; // Bersihkan hasil lama
            let sisaUtang = pinjaman;

            // Mengambil Bulan Mulai
            const [tahunMulai, bulanMulaiIndex] = bulanMulaiValue.split('-').map(Number);
            
            // Buat objek Date (bulan di JavaScript dimulai dari 0)
            let tanggalCicilan = new Date(tahunMulai, bulanMulaiIndex - 1, 1);
            
            // --- Pembuatan Baris Tabel ---
            for (let i = 1; i <= totalCicilan; i++) {
                
                // Hitung Sisa Utang
                sisaUtang -= pokokBulanan;
                
                // Format Bulan dan Tahun
                const bulanTahun = new Intl.DateTimeFormat('id-ID', { 
                    month: 'long', 
                    year: 'numeric' 
                }).format(tanggalCicilan);

                // Buat Baris Tabel
                const row = tabelBody.insertRow();
                
                // Kolom 1: Bulan Pemotongan
                const cellBulan = row.insertCell();
                cellBulan.textContent = `${i}. ${bulanTahun}`;

                // Kolom 2: Angsuran Pokok
                const cellPokok = row.insertCell();
                cellPokok.textContent = formatRupiah(pokokBulanan);

                // Kolom 3: Bunga
                const cellBunga = row.insertCell();
                cellBunga.textContent = formatRupiah(bungaBulanan);
                
                // Kolom 4: Total Cicilan
                const cellTotal = row.insertCell();
                cellTotal.textContent = formatRupiah(totalAngsuranBulanan);

                // Kolom 5: Sisa Utang (Sisa utang di akhir bulan, dibulatkan agar tidak ada -0)
                const sisaUtangBersih = Math.max(0, sisaUtang); // Jangan sampai negatif
                const cellSisa = row.insertCell();
                cellSisa.textContent = formatRupiah(sisaUtangBersih);

                // Lanjut ke bulan berikutnya
                tanggalCicilan.setMonth(tanggalCicilan.getMonth() + 1);
            }

            document.getElementById('hasilSimulasi').classList.remove('hidden');
        });

        // Fungsi untuk memformat angka menjadi Rupiah (Rp)
        const formatRupiah = (number) => {
            return new Intl.NumberFormat('id-ID', {
                style: 'currency',
                currency: 'IDR',
                minimumFractionDigits: 0
            }).format(Math.round(number)); // Membulatkan angka untuk tampilan yang bersih
        };
    </script>
